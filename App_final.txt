export default function App() {
  const [page, setPage] = useState<Page>('home');
  const [players, setPlayers] = useState<Player[]>([]);
  const [tournaments, setTournaments] = useState<Tournament[]>([]);
  const [registrations, setRegistrations] = useState<Registration[]>([]);
  const [matches, setMatches] = useState<Match[]>([]);
  const [leaderboard, setLeaderboard] = useState<LeaderboardEntry[]>([]);

  useEffect(() => {
    fetch(API_URL + '/players').then(res => res.json()).then(setPlayers);
    fetch(API_URL + '/tournaments').then(res => res.json()).then(setTournaments);
  }, []);

  return (
    <div className='min-h-screen bg-background font-inter text-foreground' style={{ scrollbarWidth: 'thin', scrollbarColor: 'rgba(0,180,255,0.2) transparent' }}>
      <Nav page={page} setPage={setPage} />
      <main className='pt-16'>
        {page === 'home' && <HomePage players={players} tournaments={tournaments} matches={matches} setPage={setPage} />}
        {page === 'players' && <PlayersPage players={players} setPlayers={setPlayers} />}
        {page === 'tournaments' && <TournamentsPage tournaments={tournaments} setTournaments={setTournaments} />}
        {page === 'registrations' && <RegistrationsPage registrations={registrations} setRegistrations={setRegistrations} players={players} tournaments={tournaments} />}
        {page === 'matches' && <MatchesPage matches={matches} setMatches={setMatches} players={players} tournaments={tournaments} />}
        {page === 'leaderboard' && <LeaderboardPage leaderboard={leaderboard} players={players} />}
        {page === 'dashboard' && <DashboardPage players={players} tournaments={tournaments} matches={matches} registrations={registrations} leaderboard={leaderboard} />}
        {page === 'login' && <LoginPage setPage={setPage} />}
      </main>
      {page !== 'login' && <Footer />}
    </div>
  );
}
